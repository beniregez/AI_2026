\section{}
The dice game is comparable to Blackjack with dice, instead of cards. Each player rolls a dice in alternating order. 
The goal is to score, in sum, a higher score than the opponent, but to not score over 21.
On each turn the players score gets added by the number of the rolled dice.
If a player scores over 21 he cannot roll the dice anymore and the other player can stop 
or roll the dice until he also scores over 21. A player wins (and the other looses) if
he stops before his score is over 21 and the other player's score is over 21 or lower than the first player's score.
The players draw if both players' score over 21.

\begin{algorithm}
\caption{ExpectiMinimax}
\begin{algorithmic}[1]
\Function{ExpectiMinimax}{$p$}
    \If{$p$ is terminal}
        \State \Return $\langle utility(p), \text{none} \rangle$
    \EndIf
    \State $best\_move := \text{none}$
    \If{$player(p) = MAX$}
        \State $v := -\infty$
    \Else
        \State $v := \infty$
    \EndIf
    \ForAll{$\langle move, p' \rangle \in \Call{Applicable}{p}$}
        \State $\langle v', best\_move' \rangle := \Call{ExpectiMinimax}{p'}$
        \State $v' = v + v' * T(p,move,p') \in \Call{Prob}{p,move}$
        \If{$(player(p)= \text{MAX} \land v' > v)$
             \textbf{or}
             $(player(p)=\text{MIN} \land v' < v)$}
            \State $v := v'$
            \State $best\_move := move$
        \EndIf
    \EndFor
    \State \Return $\langle v, best\_move \rangle$
\EndFunction
\end{algorithmic}
\end{algorithm}



\begin{algorithm}
\caption{Original Minimax from Slides}
\begin{algorithmic}[2]
\Function{Minimax}{$p$}
    \If{$p$ is terminal}
        \State \Return $\langle utility(p), \text{none} \rangle$
    \EndIf
    \State $best\_move := \text{none}$
    \If{$player(p) = MAX$}
        \State $v := -\infty$
    \Else
        \State $v := \infty$
    \EndIf
    \ForAll{$\langle move, p' \rangle \in succ(p)$}
        \State $\langle v', best\_move' \rangle := \Call{Minimax}{p'}$
        \If{$(player(p)=MAX \land v' > v)$
             \textbf{or}
             $(player(p)=MIN \land v' < v)$}
            \State $v := v'$
            \State $best\_move := move$
        \EndIf
    \EndFor
    \State \Return $\langle v, best\_move \rangle$
\EndFunction
\end{algorithmic}
\end{algorithm}
