\section{Characterizing environments}

% Characterize the following environments by describing if they are
% - static / dynamic,
% - deterministic / non-deterministic / stochastic,
% - fully / partially / not observable,
% - discrete / continuous,
% - single-agent / multi-agent
% Explain your answers.

\begin{enumerate}
    \item Monopoly
    \begin{itemize}
        \item static: The environment stays the same during the agent
        (i.e., the player) is contemplating its next action
        \item stochastic: The resulting sum of two rolled dices
        determine how many steps the active player moves forward
        \item fully observable: The player sees the entire state
        of the gameboard etc. To be precise, there are some cards
        that, if drawn by a player, only the drawing player sees and
        might keep for later use. In that case, for the other players
        the environment is only partially observable, because they
        don't see which card the other player has drawn.
        \item discrete: Only discrete quantities determine the state
        of the environment (the game), e.g. the amount of 
        available money for each player, locations of game pieces
        on the board, placed houses and hotels, event cards etc
        are discrete numbers
        \item multi-agent: Monopoly is a multiplayer game
    \end{itemize}
    \item Mars rover
    \begin{itemize}
        \item dynamic: The environment can change during the
        contemplating phase of the rover
        \item non-deterministic: The environment might always change,
        for example weather conditions like temperature or wind changes
        \item partially observable: Mars rovers can mesure a lot of
        things, but they are not capable to observe or detect
        microbes. That is, for them, the environment is not
        fully observable.
        \href{https://t3n.de/news/leben-mars-rover-sensoren-ungenau-1536159/}{Link to article}
    \end{itemize}
\end{enumerate}

\section{Evaluation of statements}
\begin{enumerate}[(a)]
    \item \textit{If all actions have cost 0, each solution for S is
    optimal.}\\
    \textbf{True}. In this case, each solution has minimal cost, and therefore
    is optimal.
    \item \textit{If $\pi = \langle a_1, ... , a_n \rangle $ is a
    minimal cost path from state $s \in S$ to state $s' \in S$ and
    $\pi' = \langle a'_1, . . . , a'_m \rangle$ is a minimal cost
    path from $s'$ to state $s'' \in S$, then $\pi'' = \langle a_1,
    ... , a_n, a'_1, ... , a'_m \rangle$ is a minimal cost path
    from $s$ to $s''$.}\\
    \textbf{False}. There might exist another path from $s$ to $s''$ with less cost
    than $\pi''$.
    \item \textit{It is possible for unit-cost state spaces (where all
    actions have cost 1) that some state has the same optimal solution
    cost as one of its successors.}\\
    \textbf{True}. Let's think of a unit-cost set space where $s_1 \in S$ and
    $s_2 \in S$.
    Further, we assume that there is an arc from $s_1$ to $s_2$.
    Addititonally, there exists arcs from the initial state to $s_1$ as well
    as $s_2$.
    Then, the cost from the initial state to $s_1$ and to $s_2$ is 1,
    thus both minimal solutions.
    At the same time, $s_2$ is a successor of $s_1$.
    \item \textit{If there is no solution for some state, then there is
    no solution for any of its predecessors.}\\
    \textbf{False}. If a state has no solution, there might still exist paths from
    the predecessors of that state to a goal state. In that case, the state
    with no solution is like a "dead" end where you end up by choosing the
    "wrong" path.
\end{enumerate}

\section{Pancake problem}
Let $\mathcal{S}$ be the state space for the pancake problem.\\
$\mathcal{S} = \langle S, A, \text{cost}, T, s_I, s_G \rangle$ with
\begin{itemize}
    \item S = $ \{ \langle s_1, s_2, ..., s_6 \rangle \mid 
    s_i \in \{ 1, ..., 6 \}, \ i \in \{1, ..., 6 \}, \
    s_i \neq s_j \ \forall \ i \neq j \}$
    \item $A = \{ \texttt{flip-1}, \ \texttt{flip-2}, ..., \ \texttt{flip-6} \}$
    \item cost(\texttt{flip-1}) = cost(\texttt{flip-2}) = ... =
    cost(\texttt{flip-6}) = 1
    \item $T$ s.t. for $i = 1, ..., 6$:\\
        $\langle
        \langle s_1, ..., s_6 \rangle,
        \texttt{flip-i},
        \langle s_1, s_{i-1}, ..., s_i, s_{i+1}, ..., s_6 \rangle
        \rangle
        \in T$
    \item $s_I = \langle 3, 2, 5, 1, 6, 4 \rangle$
    \item $s_G = \langle 1, 2, 3, 4, 5, 6 \rangle$
\end{itemize}
